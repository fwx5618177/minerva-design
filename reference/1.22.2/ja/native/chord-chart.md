# コードダイアグラム

円周上に並べたグループ間のフローを表します。各グループはフローの量に応じた長さの弧、各フローは 2 つの弧を結ぶリボンです。無向の場合、リボンはグループの組を結び、幅は両方向のフローの合計です（対称行列では各関係を 1 回だけ数えます）。そのため流入だけのグループにも弧があります。有向の場合は方向ごとに矢印付きのリボンを描きます。グループにホバーするとそのリボンを強調し、リボンにホバーすると双方向の値を表示します。凡例でグループを表示 / 非表示にでき、矢印キーでグループ、続いてリボンを順に移動します。

Package: minerva-design@1.22.2

Renderer: native

Locale: ja

### apps/docs/src/docs/pages/chord-chart/native/basic.tsx

```tsx
import { useMemo, useState } from "react";
import { useTranslation } from "react-i18next";
import { Text, View } from "react-native";
import {
  ChordChart,
  useTheme,
  type ShapeHoverEvent,
} from "minerva-design/native";

const TEAMS = ["design", "engineering", "product", "marketing"] as const;

// Messages sent in a week: row = sender, column = recipient
const MATRIX = [
  [0, 42, 28, 14],
  [36, 0, 54, 8],
  [30, 48, 0, 22],
  [12, 6, 26, 0],
];

export default function Basic() {
  const { t, i18n } = useTranslation("common", {
    keyPrefix: "docs.chord-chart",
  });
  const { colors } = useTheme();
  const [selected, setSelected] = useState<ShapeHoverEvent | null>(null);
  const groups = useMemo(
    () => TEAMS.map((team) => t(`data.teams.${team}`)),
    [t],
  );
  return (
    <View style={{ width: "100%", gap: 8 }}>
      <ChordChart
        groups={groups}
        matrix={MATRIX}
        locale={i18n.language}
        legend={false}
        onClick={setSelected}
        height={340}
        ariaLabel={t("data.teamsAria")}
      />
      <Text role="status" style={{ color: colors["text-secondary-color"] }}>
        {selected
          ? selected.source && selected.target
            ? t("native.data.flow", {
                source: selected.source,
                target: selected.target,
                value: selected.value,
                reverse: selected.reverseValue ?? 0,
              })
            : t("native.data.team", {
                name: selected.name,
                value: selected.value,
              })
          : t("native.data.tapHint")}
      </Text>
    </View>
  );
}
```
